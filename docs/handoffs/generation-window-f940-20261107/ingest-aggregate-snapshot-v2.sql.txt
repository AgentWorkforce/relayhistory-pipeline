-- SOURCE DRAFT ONLY. Not executed. Transport/projector qualification required.
-- Run preflight first through the existing qualified read-only owner path.
-- Never change the capacity collector's identity pins or grant privileges.
BEGIN READ ONLY;
SET LOCAL statement_timeout = '3s';
SET LOCAL lock_timeout = '250ms';
SELECT current_user = 'postgres' AND session_user = 'cli_login_postgres' AS expected_owner_role,
       pg_has_role(current_user, 'pg_read_all_stats', 'USAGE') AS stats_visible,
       has_table_privilege(current_user, 'sessions.ingest_tenant_slots', 'SELECT') AS slots_select,
       row_security_active('sessions.ingest_tenant_slots'::regclass) AS slots_rls_active;
ROLLBACK;

-- STOP unless the reviewed route's exact identity, target and full visibility
-- are proven. NULL/masked statistics are UNKNOWN, never healthy/idle.
-- A qualified projector admits only these closed labels and integer counts.
BEGIN READ ONLY;
SET LOCAL statement_timeout = '3s';
SET LOCAL lock_timeout = '250ms';
WITH activity AS MATERIALIZED (
  SELECT state, wait_event_type, backend_start, xact_start
  FROM pg_stat_activity WHERE usename = 'bulls96_session_ingest'
), buckets AS (
  SELECT CASE WHEN state IN ('active','idle','idle in transaction','idle in transaction (aborted)','disabled','fastpath function call') THEN state ELSE 'unknown' END AS state_bucket,
         CASE WHEN wait_event_type IN ('Activity','BufferPin','Client','Extension','IO','IPC','Lock','LWLock','Timeout') THEN wait_event_type WHEN wait_event_type IS NULL THEN 'none_or_masked' ELSE 'other' END AS wait_bucket,
         CASE WHEN backend_start IS NULL THEN 'unknown' WHEN backend_start > transaction_timestamp() THEN 'future_unknown' WHEN backend_start > transaction_timestamp()-interval '10 seconds' THEN 'lt10s' WHEN backend_start > transaction_timestamp()-interval '60 seconds' THEN '10to60s' WHEN backend_start > transaction_timestamp()-interval '5 minutes' THEN '60to300s' ELSE 'ge300s' END AS backend_age_bucket,
         CASE WHEN xact_start IS NULL THEN 'none_or_masked' WHEN xact_start > transaction_timestamp() THEN 'future_unknown' WHEN xact_start > transaction_timestamp()-interval '2 seconds' THEN 'lt2s' WHEN xact_start > transaction_timestamp()-interval '5 seconds' THEN '2to5s' WHEN xact_start > transaction_timestamp()-interval '10 seconds' THEN '5to10s' ELSE 'ge10s' END AS xact_age_bucket
  FROM activity
)
SELECT state_bucket,wait_bucket,backend_age_bucket,xact_age_bucket,count(*)::integer AS backend_count
FROM buckets GROUP BY 1,2,3,4 ORDER BY 1,2,3,4;

-- Limit 2001 bounds row materialization; overflow means sampled counts only.
-- No tenant/user/workspace/holder values, SQL text, PID or client address leave DB.
WITH slots AS MATERIALIZED (
  SELECT acquired_at,expires_at FROM sessions.ingest_tenant_slots LIMIT 2001
), buckets AS (
  SELECT CASE WHEN expires_at IS NULL THEN 'unknown' WHEN expires_at <= transaction_timestamp() THEN 'expired' ELSE 'active' END AS lease_state,
         CASE WHEN acquired_at IS NULL THEN 'unknown' WHEN acquired_at > transaction_timestamp() THEN 'future_unknown' WHEN acquired_at > transaction_timestamp()-interval '10 seconds' THEN 'lt10s' WHEN acquired_at > transaction_timestamp()-interval '60 seconds' THEN '10to60s' WHEN acquired_at > transaction_timestamp()-interval '90 seconds' THEN '60to90s' ELSE 'ge90s' END AS lease_age_bucket,
         CASE WHEN expires_at IS NULL THEN 'unknown' WHEN expires_at <= transaction_timestamp() THEN 'expired' WHEN expires_at <= transaction_timestamp()+interval '10 seconds' THEN 'lt10s' WHEN expires_at <= transaction_timestamp()+interval '60 seconds' THEN '10to60s' WHEN expires_at <= transaction_timestamp()+interval '90 seconds' THEN '60to90s' ELSE 'gt90s' END AS remaining_bucket
  FROM slots
)
SELECT lease_state,lease_age_bucket,remaining_bucket,count(*)::integer AS holder_rows,
       (SELECT count(*) > 2000 FROM slots) AS overflow_counts_are_partial
FROM buckets GROUP BY 1,2,3 ORDER BY 1,2,3;
ROLLBACK;
