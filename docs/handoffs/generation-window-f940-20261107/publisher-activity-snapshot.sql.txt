-- Source: f940 publication-runtime.ts uses application_name below for both modes.
-- Absence of activity is not proof that no idle/offline process exists.
BEGIN READ ONLY;
SET LOCAL statement_timeout = '3s';
SET LOCAL lock_timeout = '250ms';
SELECT current_user = 'postgres' AND session_user = 'cli_login_postgres' AS expected_owner_route,
       pg_has_role(current_user,'pg_read_all_stats','USAGE') AS stats_visible,
       has_table_privilege(current_user,'sessions.ingest_publication_attempts','SELECT')
       AND has_table_privilege(current_user,'sessions.ingest_publication_objects','SELECT') AS tables_visible,
       row_security_active('sessions.ingest_publication_attempts'::regclass)
       OR row_security_active('sessions.ingest_publication_objects'::regclass) AS any_rls_active;
ROLLBACK;

BEGIN READ ONLY;
SET LOCAL statement_timeout = '3s';
SET LOCAL lock_timeout = '250ms';
SELECT
 (SELECT count(*) FROM pg_stat_activity
  WHERE usename='bulls96_session_ingest' AND application_name='96bulls-canonical-publication') AS publisher_or_gc_db_backends,
 (SELECT jsonb_build_object(
    'rows',count(*),
    'created15m',count(*) FILTER(WHERE created_at>=transaction_timestamp()-interval '15 minutes'),
    'created24h',count(*) FILTER(WHERE created_at>=transaction_timestamp()-interval '24 hours'),
    'published15m',count(*) FILTER(WHERE published_at>=transaction_timestamp()-interval '15 minutes'),
    'published24h',count(*) FILTER(WHERE published_at>=transaction_timestamp()-interval '24 hours'),
    'activeUnfinishedLeases',count(*) FILTER(WHERE state IN('admitted','prepared') AND lease_until>transaction_timestamp()),
    'futureTimestamps',count(*) FILTER(WHERE created_at>transaction_timestamp() OR published_at>transaction_timestamp()))
  FROM sessions.ingest_publication_attempts) AS publication_activity,
 (SELECT jsonb_build_object(
    'rows',count(*),
    'activeGcLeases',count(*) FILTER(WHERE lease_until>transaction_timestamp()),
    'created15m',count(*) FILTER(WHERE created_at>=transaction_timestamp()-interval '15 minutes'),
    'created24h',count(*) FILTER(WHERE created_at>=transaction_timestamp()-interval '24 hours'),
    'futureCreated',count(*) FILTER(WHERE created_at>transaction_timestamp()))
  FROM sessions.ingest_publication_objects) AS publication_object_activity;
ROLLBACK;
